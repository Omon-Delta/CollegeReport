    function arrayBufferToBase64(buffer) {

      let binary = '';
      for (let i = 0; i < bytes.byteLength; i++) {

    function base64ToArrayBuffer(base64) {

            findings.push('APP1/EXIF Segment (0xFFE1)');
          }
          if (uint8Array[i] === 0xFF && uint8Array[i + 1] === 0xE2) {
            findings.push('ICC Color Profile (0xFFE2)');
          }
          if (uint8Array[i] === 0xFF && uint8Array[i + 1] === 0xED) {
            findings.push('IPTC/Photoshop Metadata (0xFFED)');
          }
        }
      }
      // Check ASCII strings for "Exif" or "http://ns.adobe.com/xap"
      const headerSample = new TextDecoder('latin1').decode(uint8Array.slice(0, 32768));
      if (headerSample.includes('Exif')) findings.push('EXIF Header Tag');
      if (headerSample.includes('http://ns.adobe.com/xap')) findings.push('Adobe XMP XML Packet');
      if (headerSample.includes('GPS')) findings.push('GPS IFD Pointer');
      return [...new Set(findings)];
    }

    async function processRealEvidenceFile(input) {
      const panel = document.getElementById('exifAuditPanel');
      const detailsEl = document.getElementById('exifAuditDetails');
      const nameEl = document.getElementById('sanitizedFileName');
      const previewControls = document.getElementById('scrubbedPreviewControls');
      const thumbImg = document.getElementById('scrubbedImgThumb');

      scrubbedAttachmentData = null;

      if (!input.files || !input.files[0]) {
        panel.classList.add('hidden');
        await updateLiveCiphertext();
        return;
      }

      const file = input.files[0];
      const originalBytes = file.size;
      const arrayBuffer = await file.arrayBuffer();
      const rawBytes = new Uint8Array(arrayBuffer);
      const safeCleanName = file.name.replace(/[^a-zA-Z0-9._-]/g, '_');

      panel.classList.remove('hidden');

      if (file.type.startsWith('image/')) {
        const detectedMarkers = detectJpegExifMarkers(rawBytes);

        // Load into HTML5 Image & redraw onto off-screen Canvas to destroy all EXIF/XMP/ICC tags
        const objectUrl = URL.createObjectURL(file);
        const img = new Image();

        img.onload = async () => {
          const canvas = document.createElement('canvas');
          canvas.width = img.naturalWidth;
          canvas.height = img.naturalHeight;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(img, 0, 0);
          URL.revokeObjectURL(objectUrl);

          const targetMime = file.type === 'image/png' ? 'image/png' : 'image/jpeg';
          canvas.toBlob(async (cleanBlob) => {
            const cleanBuffer = await cleanBlob.arrayBuffer();
            const postMarkers = detectJpegExifMarkers(new Uint8Array(cleanBuffer));
            const cleanDataUrl = await new Promise((resolve) => {
              const reader = new FileReader();
              reader.onloadend = () => resolve(reader.result);
              reader.readAsDataURL(cleanBlob);
            });

            scrubbedAttachmentData = {
              name: `scrubbed_${safeCleanName}`,
              type: targetMime,
              originalBytes: originalBytes,
              scrubbedBytes: cleanBlob.size,
              dataUrl: cleanDataUrl
            };

            nameEl.textContent = scrubbedAttachmentData.name;
            thumbImg.src = cleanDataUrl;
            previewControls.classList.remove('hidden');

            const markerSummary = detectedMarkers.length > 0
              ? `<span style="color:#FCA5A5;">${escapeHTML(detectedMarkers.join(', '))}</span>`
              : '<span style="color:#93C5FD;">None detected (re-encoded defensively anyway)</span>';

            detailsEl.innerHTML = `
              <div>• ORIGINAL FILE SIZE: <strong>${originalBytes.toLocaleString()} bytes</strong> | SCRUBBED SIZE: <strong style="color:#34D399;">${cleanBlob.size.toLocaleString()} bytes</strong></div>
              <div>• DIMENSIONS RE-ENCODED: <strong>${img.naturalWidth} × ${img.naturalHeight} px</strong> (Raw RGBA Pixel Buffer Redraw)</div>
              <div>• METADATA DETECTED BEFORE: ${markerSummary}</div>
              <div>• METADATA REMAINING AFTER: <strong style="color:#34D399;">${postMarkers.length === 0 ? '0 MARKERS (EXIF, GPS, XMP & CAMERA SERIAL PURGED)' : postMarkers.join(', ')}</strong></div>
            `;

            showToast('Image re-encoded via HTML5 Canvas; EXIF/GPS metadata destroyed.');
            await updateLiveCiphertext();
          }, targetMime, 0.90);
        };

        img.src = objectUrl;
      } else {
        // Non-image file (e.g., PDF/TXT)
        previewControls.classList.add('hidden');
        const b64 = arrayBufferToBase64(arrayBuffer);
        scrubbedAttachmentData = {
          name: `renamed_${safeCleanName}`,
          type: file.type || 'application/octet-stream',
          originalBytes: originalBytes,
          scrubbedBytes: originalBytes,
          dataUrl: `data:${file.type || 'application/octet-stream'};base64,${b64.slice(0, 120000)}`
        };

        nameEl.textContent = scrubbedAttachmentData.name;
        detailsEl.innerHTML = `
          <div>• FILE TYPE: <strong>${escapeHTML(file.type || 'Document')}</strong> (${originalBytes.toLocaleString()} bytes)</div>
          <div>• FILENAME SANITIZED: Original path stripped; renamed to <strong>${escapeHTML(scrubbedAttachmentData.name)}</strong></div>
          <div style="color:#FBBF24;">• OPSEC ADVISORY: PDFs cannot be pixel-re-encoded in-browser without rasterizing. Take a screenshot of sensitive PDF pages and upload the image so Canvas EXIF stripping can sanitize it.</div>
        `;
        await updateLiveCiphertext();
      }
    }

    function generateCsprngReceiptToken() {
      const alphabet = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
      const randomBytes = new Uint8Array(12);
      window.crypto.getRandomValues(randomBytes);
      let groups = ['', '', ''];
      for (let i = 0; i < 12; i++) {
        groups[Math.floor(i / 4)] += alphabet[randomBytes[i] % alphabet.length];
      }
      return `CR-${groups[0]}-${groups[1]}-${groups[2]}`;
    }

    async function handleReportSubmission(e) {
      e.preventDefault();

      const college = document.getElementById('repCollege').value.trim();
      const state = document.getElementById('repState').value;
      const category = document.getElementById('repCategory').value;
      const dateStr = document.getElementById('repDate').value.trim() || 'November 2026';
      const title = document.getElementById('repTitle').value.trim();
      const body = document.getElementById('repBody').value.trim();

      if (!college || !state || !title || !body) return;

      // 1. Encrypt full payload (including full scrubbed attachment) via Hybrid RSA-4096 + AES-256-GCM
      const fullPayload = {
        college,
        state,
        category,
        incidentMonth: dateStr,
        headline: title,
        narrative: body,
        attachment: scrubbedAttachmentData
          ? {
              filename: scrubbedAttachmentData.name,
              mimeType: scrubbedAttachmentData.type,
              scrubbedBytes: scrubbedAttachmentData.scrubbedBytes,
              dataUrl: scrubbedAttachmentData.dataUrl
            }
          : null,
        submittedDateBucket: new Date().toISOString().slice(0, 10)
      };

      latestEncryptedEnvelope = await hybridEncryptPayload(fullPayload);

      // 2. Generate 128-bit CSPRNG Receipt Token & immediately compute SHA-256 digest
      const plainReceiptToken = generateCsprngReceiptToken();
      const receiptSha256 = await sha256Hex(plainReceiptToken);

      // Public redacted dossier ID (decoupled from private receipt token)
      const publicCaseId = `CR-2026-${Math.floor(100 + (window.crypto.getRandomValues(new Uint16Array(1))[0] % 899))}`;

      // Store ONLY the SHA-256 digest of the receipt token in memory (zero plaintext token storage)
      reportsData.unshift({
        id: publicCaseId,
        receiptHash: receiptSha256,
        title: title,
        college: college,
        state: state,
        category: category,
        status: 'Under Review',
        date: dateStr,
        corroborations: 1,
        summary: body.length > 180 ? body.slice(0, 177) + '...' : body,
        fullText: body
      });

      const statEl = document.getElementById('statTotalReports');
      if (statEl) statEl.textContent = (1422 + reportsData.length).toLocaleString();

      applyFilters();

      // Display one-time receipt token & SHA-256 hash confirmation
      document.getElementById('generatedReceiptCode').textContent = plainReceiptToken;
      document.getElementById('generatedReceiptHashPreview').textContent = `SHA-256 DIGEST STORED: ${receiptSha256}`;
      document.getElementById('whistleblowerForm').classList.add('hidden');
      document.getElementById('receiptSuccessView').classList.remove('hidden');
    }

    function applyFilters() {
      const query = document.getElementById('searchInput').value.trim().toLowerCase();
      const stateVal = document.getElementById('stateFilter').value;
      const statusVal = document.getElementById('statusFilter').value;

      const filtered = reportsData.filter(item => {
        const matchesCat = currentCategory === 'ALL' || item.category === currentCategory;
        const matchesState = stateVal === 'ALL' || item.state === stateVal;
        const matchesStatus = statusVal === 'ALL' || item.status === statusVal;
        const matchesQuery = !query ||
          item.title.toLowerCase().includes(query) ||
          item.college.toLowerCase().includes(query) ||
          item.state.toLowerCase().includes(query) ||
          item.summary.toLowerCase().includes(query);

        return matchesCat && matchesState && matchesStatus && matchesQuery;
      });

      renderReports(filtered);
    }

    function corroborateReport(id) {
      const report = reportsData.find(r => r.id === id);
      if (!report) return;

      if (corroboratedIds.has(id)) {
        corroboratedIds.delete(id);
        report.corroborations -= 1;
        showToast(`Corroboration removed for ${id}`);
      } else {
        corroboratedIds.add(id);
        report.corroborations += 1;
        showToast(`Anonymous corroboration logged for ${id}`);
      }
      applyFilters();
      const heroCounter = document.getElementById('count-CR-2026-041');
      if (heroCounter && id === 'CR-2026-041') {
        heroCounter.textContent = report.corroborations;
      }
    }
